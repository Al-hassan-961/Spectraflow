# Spectraflow production dependencies
# ==================================
#
#   pip install -r requirements.txt
#
# This file lists ONLY what the pipeline genuinely requires, so the command
# above succeeds on every supported host, including Termux/Android. Everything
# else is an accelerator that is detected at import time -- see the commented
# sections below and the `extras_require` table in setup.py.
#
# Design rule: NumPy is the only hard runtime dependency. Every DSP stage
# (phase sanitization, clutter rejection, Butterworth filtering, STFT and the
# spectral confidence metric) has a complete pure-NumPy implementation, so the
# full pipeline runs with nothing else installed.

numpy>=1.22
uvicorn>=0.20            # pure-ASGI server; serves the application


# ---------------------------------------------------------------------------
# OPTIONAL -- not installed by default, and deliberately commented out.
#
# These are NOT installable on every platform. Leaving them uncommented makes
# `pip install -r requirements.txt` fail before it reaches them, which breaks
# the documented install for a dependency that is not actually needed.
# Install whichever you can use with the command shown next to it.
# ---------------------------------------------------------------------------

# Native DSP acceleration: builds core/ into the `spectraflow_core` extension.
# Requires a C++17 compiler. Without it the NumPy path is used, with identical
# numerical results.
#   pip install pybind11 && pip install -e .
# pybind11>=2.10

# Real neural pose inference. Without it (or without a model file) the
# deterministic CSI-driven analytic pose estimator is used instead.
#
# There is NO official onnxruntime wheel for Android/aarch64 -- the reference
# Termux deployment -- so `pip install onnxruntime` fails there by design.
#   pip install onnxruntime
# onnxruntime>=1.16

# FastAPI flavour of the server. Spectraflow serves a dependency-free ASGI
# application by default with identical routes and payloads, so this is purely
# a preference. FastAPI pulls in pydantic-core, whose Rust build fails on
# Android/aarch64 (no prebuilt wheel).
#   pip install fastapi
# fastapi>=0.100

# SciPy is not used by the pipeline at all: the Butterworth design and the IIR
# filtering are implemented directly, in both C++ and NumPy, so results are
# reproducible across backends. Listed only for downstream analysis work.
#   pip install scipy
# scipy>=1.9
